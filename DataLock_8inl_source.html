32 inline
33 bool IsCompatible( DataLockAccess a, DataLockAccess b ) noexcept
34 {
35 switch( a )
36 {
37 case DataLockAccess::None: // Any access supports no access.
38 return true;
39
40 case DataLockAccess::Readonly:
41 return b == DataLockAccess::Readonly || b == DataLockAccess::ReadWrite;
42
43 case DataLockAccess::Writeonly:
44 return b == DataLockAccess::Writeonly || b == DataLockAccess::ReadWrite;
45
46 case DataLockAccess::ReadWrite:
47 return b == DataLockAccess::ReadWrite;
48
49 default:
50 return false;
51 }
52 }

289 template < typename T >
290 T * DataLock<T>::GetItem( size_t index )
291 {
292 if (index >= m_count)
293 {
294 throw unify::Exception( "Attempted to access index out of range! (" + *unify::ToString( index ) + " to " + *unify::Cast< std::string >( m_count ) + ")" );
295 }
296
297 if ( !HasReadAccess( m_type ) || !HasWriteAccess( m_type ) )
298 {
299 throw unify::Exception( "Attempted to access " + DataLockAccess::ToString( m_type ) + " for " + DataLockAccess::ToString( DataLockAccess::ReadWrite ) + " access!" );
300 }
301
302 return m_data + m_stride * index;
