195 template< typename T >
196 std::optional< RayHit< T > > BBox< T >::Intersects( const Ray< T > & ray ) const
197 {
198 T t_min = -std::numeric_limits< T >::max();
199 T t_max = std::numeric_limits< T >::max();
200 int hit_axis = -1;
201 T hit_side = ( T )1;
202
203 for ( int i = 0; i < 3; ++i )
204 {
205 if ( fabsf( ray.direction[i] ) < std::numeric_limits< T >::epsilon() )
206 {
207 if ( ray.origin[i] < inf[i] || ray.origin[i] > sup[i] ) return std::nullopt;
208 }
209 else
210 {
211 T ood = ( T )1 / ray.direction[i];
212 T t1 = ( inf[i] - ray.origin[i] ) * ood;
213 T t2 = ( sup[i] - ray.origin[i] ) * ood;
214
215 T side = ( T )-1;
216 if ( t1 > t2 )
217 {
218 std::swap( t1, t2 );
219 side = ( T )1;
220 }
221
222 if ( t1 > t_min )
223 {
224 t_min = t1;
225 hit_axis = i;
226 hit_side = side;
227 }
228
229 if ( t2 < t_max ) t_max = t2;
230
231 if ( t_min > t_max ) return std::nullopt;
232 }
233 }
234
235 if ( t_max < ( T )0 ) return std::nullopt;
236
237 RayHit< T > hit;
238 hit.distance = ( t_min < ( T )0 ) ? ( T )0 : t_min;
239 hit.point = ray.origin + ( ray.direction * hit.distance );
240
241 hit.normal = V3< T >( ( T )0, ( T )0, ( T )0 );
242 if ( hit_axis != -1 )
243 {
244 hit.normal[hit_axis] = hit_side;
245 }
246
247 return hit;
248 }

250 template< typename T >
251 std::optional< LineSegment3< T > > BBox< T >::Clip( const Ray< T > & ray ) const
252 {
253 T t_min = -std::numeric_limits< T >::max();
254 T t_max = std::numeric_limits< T >::max();
255
256 for ( int i = 0; i < 3; ++i )
257 {
258 if ( fabsf( ray.direction[i] ) < std::numeric_limits< T >::epsilon() )
259 {
260 if ( ray.origin[i] < inf[i] || ray.origin[i] > sup[i] ) return std::nullopt;
261 }
262 else
263 {
264 T ood = ( T )1 / ray.direction[i];
265 T t1 = ( inf[i] - ray.origin[i] ) * ood;
266 T t2 = ( sup[i] - ray.origin[i] ) * ood;
267
268 if ( t1 > t2 ) std::swap( t1, t2 );
269
270 if ( t1 > t_min ) t_min = t1;
271 if ( t2 < t_max ) t_max = t2;
272
273 if ( t_min > t_max ) return std::nullopt;
274 }
275 }
276
277 // If t_max is negative, the entire box is behind the ray origin.
278 if ( t_max < ( T )0 )
279 {
280 return std::nullopt;
281 }
282
283 // Clamp t_min to 0 if the ray origin is inside the box.
284 T start_t = ( t_min < ( T )0 ) ? ( T )0 : t_min;
285
286 return LineSegment3< T >(
287 ray.origin + ( ray.direction * start_t ),
288 ray.origin + ( ray.direction * t_max )
289 );
290 }

292 template< typename T >
293 bool BBox< T >::Intersects( const Ray< T > & ray, T t0, T t1 ) const
294 {
295 T t_min = t0;
296 T t_max = t1;
297
298 for ( int i = 0; i < 3; ++i )
299 {
300 if ( fabsf( ray.direction[i] ) < std::numeric_limits< T >::epsilon() )
301 {
302 // Ray is parallel to slab; if origin is outside, no intersection possible
303 if ( ray.origin[i] < inf[i] || ray.origin[i] > sup[i] ) return false;
304 }
305 else
306 {
307 T ood = ( T )1 / ray.direction[i];
308 T t_near = ( inf[i] - ray.origin[i] ) * ood;
309 T t_far = ( sup[i] - ray.origin[i] ) * ood;
310
311 if ( t_near > t_far ) std::swap( t_near, t_far );
312
313 // Constrain the interval to the overlapping region
314 if ( t_near > t_min ) t_min = t_near;
315 if ( t_far < t_max ) t_max = t_far;
316
317 // If the interval becomes invalid, the ray misses the box
318 if ( t_min > t_max ) return false;
319 }
320 }
321
322 return true;
323 }

392 template< typename T >
393 BSphere< T > BBox< T >::MakeBSphere() const
394 {
395 // The center of the sphere is the center of the box
396 V3< T > center = ( inf + sup ) * ( T )0.5;
397
398 // The radius is the distance from the center to one of the corners (like sup)
