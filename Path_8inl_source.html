119 inline
120 bool Path::operator==( const Path & path ) const
121 {
122 if ( m_path.length() != path.m_path.length() )
123 {
124 return false;
125 }
126
127 unsigned int i = 0;
128 while( i < m_path.length() )
129 {
130 char l = m_path[ i ];
131 char r = path.m_path[ i ];
132 if ( l == '\\' )
133 {
134 l = '/';
135 }
136 if ( r == '\\' )
137 {
138 r = '/';
139 }
140
141 if ( l != r )
142 {
143 return false;
144 }
145 ++i;
146 }
147
148 return true;
149 }

172 std::vector< std::string > Path::Split() const
173 {
174 std::vector< std::string > parts;
175 size_t start = 0;
176 size_t p;
177 std::string asString = ToString();
178 for ( p = 0; p != asString.size(); ++p )
179 {
180 if (asString.at( p ) == '\\' || asString.at( p ) == '/' )
181 {
182 std::string part;
183 if ( p != 0 )
184 {
185 part = asString.substr( start, p - start );
186 }
187
188 part += "/";
189 parts.push_back( part );
190 start = p + 1;
191 }
192 }
193 if ( p != 0 && p != start )
194 {
195 parts.push_back( asString.substr( start, p - start ) );
196 start = p + 1;
197 }
198 return parts;
199 }

212 inline
213 Path & Path::Normalize()
214 {
215 std::vector< std::string > parts = Split();
216 for ( std::vector< std::string >::const_iterator itr = parts.begin(); itr != parts.end(); ++itr )
217 {
218 if ( itr != parts.begin() && String::BeginsWith( *itr, ".." ) )
219 {
220 itr = parts.erase( itr - 1 ); // Erase what ever leads.
221 itr = parts.erase( itr ); // Erase the "..".
222 }
223 else if ( itr != parts.begin() && String::BeginsWith( *itr, "." ) )
224 {
225 itr = parts.erase( itr ); // Erase the ".".
226 }
227 }
228 return Join( parts );
229 }

320 inline
321 Path & Path::Combine( const Path & left, const Path & right )
322 {
323 // If one or the other is empty, return the other.
324 if ( left.m_path.empty() || left.m_path == "file:///" )
325 {
326 m_path = right.ToXPath();
327 return *this;
328 }
329 else if ( right.m_path.empty() || right.m_path == "file:///" )
330 {
331 m_path = left.ToXPath();
332 return *this;
333 }
334
335 std::string pathOut( left.m_path ); // Create a new path instead of working directly with ours incase L or R is us.
336
337 // Prefix if needed.
338 char lastChar = String::RightString( left.ToString(), 1 )[ 0 ];
339 char firstChar = right.ToString()[ 0 ];
340 if ( lastChar != '\\' && lastChar != '/' && firstChar != '\\' && firstChar != '/' )
341 {
342 pathOut += '/' + right.ToString();
343 }
344 else if ( ( lastChar == '\\' || lastChar == '/' ) && ( firstChar == '\\' || firstChar == '/' ) )
345 {
346 // Remove prefix if it exists with both
347 pathOut += right.ToString().substr( 1 );
348 }
349 else
350 {
351 pathOut += right.ToString();
352 }
353
354 m_path = pathOut;
355 return *this;
356 }

378 inline
379 void Path::ChangeExtension( std::string extension )
380 {
381 std::string pathOut;
382 size_t i = m_path.find_last_of( "./\\" );
383
384 // Check if we found an extension
385 if ( i == std::string::npos || m_path[ i ] == '/' || m_path[ i ] == '\\' )
386 {
387 pathOut = m_path;
388 }
389 else
390 {
391 pathOut = m_path.substr( 0, i );
392 }
393
394 // Allow an extension to be removed
395 if ( ! extension.empty() )
396 {
397 pathOut += ( extension.at( 0 ) == '.' ? "" : "." ) + extension;
398 }
399
400 m_path = pathOut;
401 }
